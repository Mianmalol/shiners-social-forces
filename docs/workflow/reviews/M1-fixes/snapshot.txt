"""Descriptive pilot measurements defined in docs/methods.md.

Inputs are one trial. Pixels use image coordinates (y down). Undefined quantities
remain NaN; functions do not impute missing samples or infer behavioral responses.
"""

from __future__ import annotations

import numpy as np
import pandas as pd
from scipy.signal import savgol_filter


def _single_trial(tracks):
    for column in ("group", "sequence"):
        if column in tracks and tracks[column].nunique(dropna=False) > 1:
            raise ValueError("Metrics require one trial at a time")
    if tracks.empty:
        raise ValueError("Empty tracks")
    if tracks.duplicated(["frame", "id"]).any():
        raise ValueError("Duplicate track keys")


def _scale(length_scale):
    if not np.isfinite(length_scale) or length_scale <= 0:
        raise ValueError("Length scale must be finite and positive")


def _dense(tracks, columns):
    """A frame-by-fish view with explicit identical fish coverage."""
    _single_trial(tracks)
    frames = np.sort(tracks.frame.unique())
    fish = np.sort(tracks.id.unique())
    ordered = tracks.sort_values(["frame", "id"])
    if len(ordered) != len(frames) * len(fish):
        raise ValueError("Incomplete fish set")
    # Unique keys plus the full Cartesian cardinality ensures every fish/frame.
    arrays = {column: ordered[column].to_numpy(dtype=float).reshape(len(frames), len(fish))
              for column in columns}
    return frames, fish, ordered, arrays


def kinematics(tracks, window=11, order=3):
    """Return original rows/index plus segment- and gap-safe temporal derivatives.

    vx/vy: pixels/s; ax/ay: pixels/s²; turn_rate: radians/s (clockwise positive).
    Discard half a smoothing window plus one sample at both block ends, so the
    known noisier first/last segment samples do not enter retained estimates.
    """
    _single_trial(tracks)
    if not isinstance(window, (int, np.integer)) or window < 3 or window % 2 != 1:
        raise ValueError("Window must be an odd integer >= 3")
    if not isinstance(order, (int, np.integer)) or order < 2 or order >= window:
        raise ValueError("Polynomial order must be >= 2 and below window")
    if not np.isfinite(tracks[["time", "frame", "X", "Y", "ANGLE"]].to_numpy(dtype=float)).all():
        raise ValueError("Nonfinite kinematics input")
    result = tracks.copy()
    output = {name: np.full(len(tracks), np.nan) for name in ["xs", "ys", "vx", "vy", "ax", "ay", "turn_rate"]}
    jump = np.zeros(len(tracks), dtype=bool)
    heading_valid = np.ones(len(tracks), dtype=bool)
    half = window // 2
    trim = half + 1
    # groupby.indices uses positional indexes, preserving even nonunique row labels.
    for positions in tracks.groupby(["id", "segment"], sort=False).indices.values():
        positions = positions[np.argsort(tracks.frame.to_numpy()[positions], kind="stable")]
        frame = tracks.frame.to_numpy()[positions]
        for block in np.split(positions, np.flatnonzero(np.diff(frame) != 1) + 1):
            if len(block) < 2:
                continue
            sample = tracks.iloc[block]
            heading = sample.ANGLE.to_numpy(dtype=float)
            step = (np.diff(heading) + np.pi) % (2*np.pi) - np.pi
            flagged = np.flatnonzero(np.abs(step) > np.pi/2)
            block_jump = np.zeros(len(block), dtype=bool)
            block_jump[flagged] = True
            block_jump[flagged+1] = True
            jump[block] = block_jump
            contaminated = np.convolve(np.pad(block_jump, (half, half)), np.ones(window), mode="valid") > 0
            heading_valid[block] = ~contaminated
            delta = np.diff(sample.time.to_numpy(dtype=float))
            dt = float(np.median(delta))
            if dt <= 0 or not np.allclose(delta, dt, rtol=1e-6, atol=1e-10):
                raise ValueError("Irregular or nonpositive time increments within a continuous block")
            if len(block) <= 2*trim:
                continue
            interior = block[trim:-trim]
            for coordinate, velocity, acceleration in [("X", "vx", "ax"), ("Y", "vy", "ay")]:
                values = sample[coordinate].to_numpy(dtype=float)
                smooth = savgol_filter(values, window, order, mode="interp")
                output["xs" if coordinate == "X" else "ys"][interior] = smooth[trim:-trim]
                for derivative, name in [(1, velocity), (2, acceleration)]:
                    fitted = savgol_filter(values, window, order, deriv=derivative, delta=dt, mode="interp")
                    output[name][interior] = fitted[trim:-trim]
            turn = savgol_filter(np.unwrap(heading), window, order, deriv=1, delta=dt, mode="interp")
            output["turn_rate"][interior] = turn[trim:-trim]
    output["turn_rate"][~heading_valid] = np.nan
    for name, values in output.items():
        result[name] = values
    result["heading_jump"] = jump
    result["heading_valid"] = heading_valid
    axis_error = np.full(len(tracks), np.nan)
    if all(column in tracks for column in ["poseX0", "poseY0", "poseX20", "poseY20"]):
        dx = (tracks.poseX0-tracks.poseX20).to_numpy(dtype=float)
        dy = (tracks.poseY0-tracks.poseY20).to_numpy(dtype=float)
        valid_axis = np.isfinite(dx) & np.isfinite(dy) & (np.hypot(dx, dy) > 1e-8)
        axis_error[valid_axis] = ((tracks.ANGLE.to_numpy()[valid_axis] - np.arctan2(dy[valid_axis], dx[valid_axis]) + np.pi)
                                  % (2*np.pi) - np.pi)
    result["heading_axis_error"] = axis_error
    return result


def curvature(tracks, neighborhood=7):
    """Signed midpoint curvature in 1/pixel from a local cubic midline fit.

    Equal arc-spacing coordinates are centered on keypoint 10. Points 21 and 22
    (eyes) are never used. Positive sign is clockwise head-to-tail bending.
    """
    if (not isinstance(neighborhood, (int, np.integer)) or neighborhood < 5
            or neighborhood > 21 or neighborhood % 2 != 1):
        raise ValueError("Curvature neighborhood must be odd, between 5 and 21")
    indices = np.arange(10-neighborhood//2, 11+neighborhood//2)
    u = indices / 20.0 - 0.5
    design = np.column_stack([np.ones(neighborhood), u, u*u, u*u*u])
    inverse = np.linalg.pinv(design)
    x = tracks[[f"poseX{i}" for i in indices]].to_numpy(dtype=float)
    y = tracks[[f"poseY{i}" for i in indices]].to_numpy(dtype=float)
    dx, dy = x @ inverse[1], y @ inverse[1]
    ddx, ddy = 2 * (x @ inverse[2]), 2 * (y @ inverse[2])
    speed = np.hypot(dx, dy)
    valid = np.isfinite(x).all(axis=1) & np.isfinite(y).all(axis=1) & (speed > 1e-8)
    out = np.full(len(tracks), np.nan)
    out[valid] = (dx[valid]*ddy[valid] - dy[valid]*ddx[valid]) / speed[valid]**3
    return out


def collective(tracks_with_kinematics, length_scale, low_speed=.5, min_fraction=.8):
    """Per-frame heading alignment, spread and centroid-relative circulation."""
    _scale(length_scale)
    if not np.isfinite(low_speed) or low_speed <= 0 or not 0 < min_fraction <= 1:
        raise ValueError("Low speed must be positive and min_fraction in (0,1]")
    x_col, y_col = ("xs", "ys") if {"xs", "ys"}.issubset(tracks_with_kinematics.columns) else ("X", "Y")
    frames, fish, ordered, a = _dense(tracks_with_kinematics, [x_col, y_col, "ANGLE", "vx", "vy"])
    position = np.stack([a[x_col], a[y_col]], axis=2)
    velocity = np.stack([a["vx"], a["vy"]], axis=2)
    if not np.isfinite(a["ANGLE"]).all():
        raise ValueError("Nonfinite collective heading")
    relative = position - position.mean(axis=1, keepdims=True)
    radius = np.linalg.norm(relative, axis=2)
    centroid_v = velocity.mean(axis=1)
    all_velocity = np.isfinite(velocity).all(axis=(1, 2))
    all_geometry = np.isfinite(position).all(axis=(1, 2))
    all_valid = all_velocity & all_geometry
    relative_v = velocity - centroid_v[:, None, :]
    relative_speed = np.linalg.norm(relative_v, axis=2)
    eligible = ((radius > 1e-6*length_scale) & (relative_speed >= low_speed*length_scale)
                & all_valid[:, None])
    fractions = eligible.mean(axis=1)
    # Eligibility is unknown when the centroid-relative velocities cannot be computed.
    fractions = np.where(all_valid, fractions, np.nan)
    cross = relative[:, :, 0]*relative_v[:, :, 1] - relative[:, :, 1]*relative_v[:, :, 0]
    normalized = np.zeros_like(cross)
    np.divide(cross, radius*relative_speed, out=normalized, where=eligible)
    counts = eligible.sum(axis=1)
    circulation = np.full(len(frames), np.nan)
    signed = np.full(len(frames), np.nan)
    reference = np.full(len(frames), np.nan)
    valid = all_valid & (fractions >= min_fraction) & (counts > 0)
    signed[valid] = np.clip(normalized[valid].sum(axis=1)/counts[valid], -1, 1)
    circulation[valid] = np.abs(signed[valid])
    reference[valid] = np.sqrt(1/(np.pi*counts[valid]))
    # A little rounding can make an analytic unit circulation exceed one by epsilon.
    circulation[valid] = np.clip(circulation[valid], 0, 1)
    centroid_speed = np.linalg.norm(centroid_v, axis=1)/length_scale
    centroid_speed[~all_velocity] = np.nan
    first = ordered.groupby("frame", sort=True)[["time", "segment"]].first()
    polarization = np.hypot(np.cos(a["ANGLE"]).mean(axis=1), np.sin(a["ANGLE"]).mean(axis=1))
    if "heading_valid" in ordered:
        polarization[~ordered.heading_valid.to_numpy(dtype=bool).reshape(len(frames), len(fish)).all(axis=1)] = np.nan
    return pd.DataFrame({
        "frame": frames, "time": first.time.to_numpy(), "segment": first.segment.to_numpy(),
        "polarization": polarization,
        "spread_bl": np.sqrt((radius**2).mean(axis=1))/length_scale,
        "centroid_speed_bl_s": centroid_speed, "circulation": circulation,
        "eligible_fraction": fractions, "signed_circulation": signed,
        "iid_direction_reference": reference,
    })


def spatial(tracks, length_scale, extent=10, bins=40):
    """All-neighbor occupancy in focal forward/right coordinates.

    histogram[forward_bin, right_bin] contains raw ordered-pair counts. density
    divides by all focal observations and bin area in square body lengths.
    """
    _scale(length_scale)
    if not np.isfinite(extent) or extent <= 0 or not isinstance(bins, (int, np.integer)) or bins < 1:
        raise ValueError("Positive extent and integer bin count required")
    frames, fish, _, a = _dense(tracks, ["X", "Y", "ANGLE"])
    if not all(np.isfinite(values).all() for values in a.values()):
        raise ValueError("Nonfinite spatial inputs")
    edges = np.linspace(-extent, extent, bins+1)
    counts = np.zeros((bins, bins), dtype=float)
    nearest = np.full((len(frames), len(fish)), np.nan)
    off_diagonal = ~np.eye(len(fish), dtype=bool)
    # Chunk frame dimension so larger schools do not allocate a trial-sized NxN tensor.
    for start in range(0, len(frames), 128):
        stop = min(start+128, len(frames))
        x, y, theta = (a[column][start:stop] for column in ["X", "Y", "ANGLE"])
        dx = (x[:, None, :] - x[:, :, None])/length_scale
        dy = (y[:, None, :] - y[:, :, None])/length_scale
        distance = np.hypot(dx, dy)
        distance[:, np.arange(len(fish)), np.arange(len(fish))] = np.inf
        if len(fish) > 1:
            nearest[start:stop] = distance.min(axis=2)
        cosine, sine = np.cos(theta)[:, :, None], np.sin(theta)[:, :, None]
        forward = (dx*cosine + dy*sine)[:, off_diagonal].ravel()
        right = (-dx*sine + dy*cosine)[:, off_diagonal].ravel()
        counts += np.histogram2d(forward, right, bins=(edges, edges))[0]
    focal_count = len(frames)*len(fish)
    pairs = focal_count*(len(fish)-1)
    bin_area = (edges[1]-edges[0])**2
    return {
        "nearest": pd.DataFrame({"frame": np.repeat(frames, len(fish)), "id": np.tile(fish, len(frames)),
                                 "distance_bl": nearest.ravel()}),
        "histogram": counts, "density": counts/(focal_count*bin_area),
        "bin_edges": edges, "focal_count": focal_count,
        "shown_pair_fraction": float(counts.sum()/pairs) if pairs else float("nan"),
    }


def visibility(tracks, network):
    """Observer out-degree including zeros, restricted to covered frames."""
    _single_trial(tracks)
    if network.duplicated(["frame", "id1", "id2"]).any():
        raise ValueError("Duplicate visibility edges")
    if (network.id1 == network.id2).any():
        raise ValueError("Self visibility edges")
    track_keys = pd.MultiIndex.from_frame(tracks[["frame", "id"]])
    for column in ["id1", "id2"]:
        if not pd.MultiIndex.from_frame(network[["frame", column]]).isin(track_keys).all():
            raise ValueError("Dangling visibility endpoint")
    covered = tracks[tracks.frame.isin(network.frame.unique())].sort_values(["frame", "id"])
    keys = pd.MultiIndex.from_frame(covered[["frame", "id"]])
    counts = network.groupby(["frame", "id1"]).size().reindex(keys, fill_value=0)
    degree = covered[["frame", "id"]].reset_index(drop=True).copy()
    degree["degree"] = counts.to_numpy(dtype=int)
    per_frame = degree.groupby("frame", as_index=False).agg(mean_degree=("degree", "mean"))
    pairs = pd.MultiIndex.from_frame(network[["frame", "id1", "id2"]])
    reverse = pd.MultiIndex.from_arrays([network.frame, network.id2, network.id1])
    reciprocity = float(reverse.isin(pairs).mean()) if len(network) else float("nan")
    return {"degree": degree, "per_frame": per_frame, "reciprocal_edge_fraction": reciprocity}


def heading_diagnostics(k, length_scale, low_speed=.5):
    """Describe jumps and sustained travel opposite heading, without exclusion.

    Reverse motion can be real backward movement, a tracking error, or another
    mismatch. These diagnostics do not identify proven head-tail flips. The
    velocity eligibility denominator is computed before any heading jump mask.
    """
    _single_trial(k)
    _scale(length_scale)
    if not np.isfinite(low_speed) or low_speed <= 0:
        raise ValueError("Low-speed threshold must be finite and positive")
    angle = k.ANGLE.to_numpy(dtype=float)
    vx, vy = k.vx.to_numpy(dtype=float), k.vy.to_numpy(dtype=float)
    if not np.isfinite(angle).all():
        raise ValueError("Nonfinite heading diagnostic inputs")
    speed = np.hypot(vx, vy)/length_scale
    eligible = np.isfinite(speed) & (speed >= low_speed)
    slip = (np.arctan2(vy, vx) - angle + np.pi) % (2*np.pi) - np.pi
    reverse = eligible & (np.abs(slip) > np.pi/2)
    unmasked = k.heading_valid.to_numpy(dtype=bool) if "heading_valid" in k else np.ones(len(k), dtype=bool)
    frames = k.frame.to_numpy()
    events = []
    best = {int(fish): {"id": int(fish), "start": None, "end": None, "segment": None, "n": 0}
            for fish in np.sort(k.id.unique())}
    for (fish, segment), positions in k.groupby(["id", "segment"], sort=True).indices.items():
        positions = positions[np.argsort(frames[positions], kind="stable")]
        block_frames = frames[positions]
        step = (np.diff(angle[positions]) + np.pi) % (2*np.pi) - np.pi
        jumps = np.flatnonzero((np.diff(block_frames) == 1) & (np.abs(step) > np.pi/2))
        for index in jumps:
            events.append({"id": int(fish), "segment": int(segment),
                           "prev_frame": int(block_frames[index]), "frame": int(block_frames[index+1]),
                           "signed_wrapped_step": float(step[index])})
        # Retain only reverse samples. A false/ineligible sample or frame gap
        # breaks a run even if reverse motion resumes on the next stored row.
        indices = np.flatnonzero(reverse[positions])
        if not len(indices):
            continue
        cuts = np.flatnonzero((np.diff(indices) != 1) | (np.diff(block_frames[indices]) != 1))+1
        for run in np.split(indices, cuts):
            record = {"id": int(fish), "start": int(block_frames[run[0]]),
                      "end": int(block_frames[run[-1]]), "segment": int(segment), "n": int(len(run))}
            old = best[int(fish)]
            if record["n"] > old["n"] or (record["n"] == old["n"] and record["start"] < old["start"]):
                best[int(fish)] = record
    denominator = int(eligible.sum())
    return {
        "low_speed_bl_s": float(low_speed), "heading_events": events,
        "speed_eligible_rows_before_heading_mask": denominator,
        "reverse_motion_rows": int(reverse.sum()),
        "reverse_motion_fraction": float(reverse.sum()/denominator) if denominator else None,
        "reverse_motion_rows_outside_jump_mask": int((reverse & unmasked).sum()),
        "longest_reverse_runs_per_fish": list(best.values()),
        "interpretation": "Flags reverse travel relative to tracked heading, not proven head-tail flips. "
                          "Sustained intervals are not automatically excluded without video validation.",
        "mask_policy": "Finite speed >= threshold; heading_valid is not used in the primary denominator or runs.",
    }

"""Synthetic checks derived from analytic geometry, not real-data expectations."""

import unittest
import json
import numpy as np
import pandas as pd

from shiners.metrics import kinematics, curvature, collective, spatial, visibility, heading_diagnostics


def trajectory(n=61, dt=.1):
    t = np.arange(n)*dt
    return pd.DataFrame({"frame": np.arange(n), "id": 0, "segment": 0,
                         "time": t, "X": 3*t, "Y": 4*t, "ANGLE": 0.})


def frame(x, y, theta=None, vx=None, vy=None):
    n = len(x)
    return pd.DataFrame({"frame": 0, "time": 0., "segment": 0, "id": np.arange(n),
                         "X": x, "Y": y, "ANGLE": np.zeros(n) if theta is None else theta,
                         "vx": np.zeros(n) if vx is None else vx, "vy": np.zeros(n) if vy is None else vy})


def midline(x, y):
    return pd.DataFrame({**{f"poseX{i}": [v] for i, v in enumerate(x)},
                         **{f"poseY{i}": [v] for i, v in enumerate(y)},
                         "poseX21": [1e9], "poseY21": [-1e9]})


class KinematicsTests(unittest.TestCase):
    def test_constant_velocity_and_preserved_index(self):
        tr = trajectory().sample(frac=1, random_state=1)
        out = kinematics(tr)
        self.assertTrue(out.index.equals(tr.index))
        valid = out.frame.between(6, 54)
        np.testing.assert_allclose(out.loc[valid, "vx"], 3, atol=1e-10)
        np.testing.assert_allclose(out.loc[valid, "vy"], 4, atol=1e-10)
        np.testing.assert_allclose(out.loc[valid, "ax"], 0, atol=1e-9)
        self.assertTrue(out.loc[~valid, ["vx", "vy", "ax", "ay", "turn_rate"]].isna().all().all())

    def test_acceleration_and_wrapped_signed_turn(self):
        tr = trajectory()
        tr["X"] = 2*tr.time**2 + 3*tr.time
        tr["ANGLE"] = (3 + .7*tr.time + np.pi) % (2*np.pi) - np.pi
        out = kinematics(tr)
        np.testing.assert_allclose(out.ax.dropna(), 4, atol=1e-9)
        np.testing.assert_allclose(out.turn_rate.dropna(), .7, atol=1e-10)

    def test_gaps_and_segment_boundaries_never_bridge(self):
        tr = trajectory(100)
        tr = tr[tr.frame != 40].copy()
        tr.loc[tr.frame > 40, "X"] += 1000
        tr.loc[tr.frame >= 70, "segment"] = 1
        out = kinematics(tr)
        for center in [40, 70]:
            # Extra sample removes noisy block-edge data from retained windows.
            affected = out.frame.between(center-6, center+5)
            self.assertTrue(out.loc[affected, "vx"].isna().all())
        np.testing.assert_allclose(out.vx.dropna(), 3, atol=1e-9)

    def test_irregular_sampling_is_rejected_even_short_block(self):
        tr = trajectory(8)
        tr.loc[4, "time"] += .01
        with self.assertRaisesRegex(ValueError, "Irregular"):
            kinematics(tr)

    def test_heading_jump_contaminates_only_heading_metrics_and_sg_support(self):
        tr = trajectory()
        tr.loc[tr.frame >= 30, "ANGLE"] = np.pi
        tr["poseX0"] = tr.X+1; tr["poseX20"] = tr.X-1
        tr["poseY0"] = tr.Y; tr["poseY20"] = tr.Y
        out = kinematics(tr)
        self.assertEqual(out.loc[out.heading_jump, "frame"].tolist(), [29, 30])
        self.assertEqual(out.loc[~out.heading_valid, "frame"].tolist(), list(range(24, 36)))
        self.assertTrue(out.loc[~out.heading_valid, "turn_rate"].isna().all())
        self.assertTrue(out.loc[~out.heading_valid, "vx"].notna().all())
        self.assertTrue(out.heading_valid.iloc[0])  # A kinematic edge is not a bad heading.
        self.assertAlmostEqual(abs(out.heading_axis_error.iloc[40]), np.pi)
        organized = collective(out, 1)
        self.assertTrue(np.isnan(organized.polarization.iloc[29]))
        self.assertEqual(organized.polarization.iloc[0], 1)


class CurvatureTests(unittest.TestCase):
    def test_straight_line_and_degenerate_curve(self):
        np.testing.assert_allclose(curvature(midline(np.linspace(0, 20, 21), np.zeros(21))), 0, atol=1e-12)
        self.assertTrue(np.isnan(curvature(midline(np.ones(21), np.ones(21)))[0]))

    def test_circular_arc_sign_and_translation(self):
        angle = np.linspace(-.3, .3, 21)
        # Increasing angle on an image-coordinate circle is clockwise.
        for radius in [10., 100.]:
            tr = midline(1000+radius*np.cos(angle), 2000+radius*np.sin(angle))
            self.assertAlmostEqual(curvature(tr)[0], 1/radius, delta=1e-3/radius)
            reflected = midline(1000+radius*np.cos(angle), 2000-radius*np.sin(angle))
            self.assertAlmostEqual(curvature(reflected)[0], -1/radius, delta=1e-3/radius)


class CollectiveTests(unittest.TestCase):
    def test_parallel_translation_is_aligned_but_circulation_undefined(self):
        tr = frame([-1, 1], [0, 0], vx=[3, 3], vy=[4, 4])
        out = collective(tr, 2).iloc[0]
        self.assertEqual(out.polarization, 1)
        self.assertAlmostEqual(out.spread_bl, .5)
        self.assertAlmostEqual(out.centroid_speed_bl_s, 2.5)
        self.assertTrue(np.isnan(out.circulation))
        self.assertEqual(out.eligible_fraction, 0)

    def test_circular_motion_and_translation_invariance(self):
        a = np.arange(8)*2*np.pi/8
        tr = frame(5*np.cos(a), 5*np.sin(a), theta=a+np.pi/2,
                   vx=-2*np.sin(a), vy=2*np.cos(a))
        out = collective(tr, 1).iloc[0]
        self.assertAlmostEqual(out.circulation, 1)
        self.assertAlmostEqual(out.signed_circulation, 1)
        self.assertAlmostEqual(out.iid_direction_reference, np.sqrt(1/(8*np.pi)))
        self.assertAlmostEqual(out.polarization, 0, delta=1e-12)
        tr["vx"] += 20; tr["vy"] -= 7
        self.assertAlmostEqual(collective(tr, 1).circulation.iloc[0], 1)
        tr["vx"] *= -1; tr["vy"] *= -1
        reversed_out = collective(tr, 1).iloc[0]
        self.assertAlmostEqual(reversed_out.signed_circulation, -1)
        self.assertAlmostEqual(reversed_out.circulation, 1)

    def test_invalid_velocity_and_insufficient_eligibility(self):

        self.assertTrue(np.isnan(out.centroid_speed_bl_s))
        self.assertTrue(np.isnan(out.eligible_fraction))

        out = collective(tr, 1).iloc[0]

        self.assertTrue(np.isnan(out.circulation))

# Golden shiners: first descriptive atlas

**Scope:** one 10-fish trial, `0066`, approximately 13.7 minutes. This is an exploratory pilot, not a completed all-trial study. No social-force model, leadership analysis or causal inference has been fitted.

OpenAI implemented and executed the analysis. Claude Opus 5.5 reviewed the methods and requested corrections before the real run. Implementation-review status is tracked in [workflow status](../docs/workflow/status.md). Tool-free Claude review is independent reasoning, not independent execution. The strict reproduction/release gate remains pending.

## What we can trust so far

The tables contain 245,780 fish-frame observations, 24,578 tracked frames and 1,662,869 directed visibility links. Required values are finite, keys are unique, each frame contains all ten fish, and all network endpoints match tracks. Stored network distances agree with recomputed center distances to numerical precision. Seven recording segments are separated by six missing frames; derivatives do not bridge them.

Four fish-frame rows were flagged around two large heading changes. Their smoothing support was excluded from heading-derived results. No head-to-tail axis disagreement exceeded 90 degrees. This is a QA check, not proof that all identities/headings are correct. The two events occur on **different fish**: fish 2 at frame 18174 (+1.710 rad), and fish 1 at frame 20279 (+1.892 rad), both in segment 5. They do not bracket a flip/unflip interval on one fish. Separately, **266 of 222,181 speed-eligible rows (0.120%)** have travel direction more than 90 degrees from recorded heading. The longest such run is fish 3, frames 15808–15830 (23 frames). These unresolved candidates may reflect backward motion or tracking/derivative defects; axis agreement does not rule out simultaneous head/pose reversal. No automatic heading correction was applied. The exact events and longest run per fish are retained in summary.json, and these intervals require targeted video review before strong heading-based interpretations.

The shared descriptive scale is median body length **38.99 pixels**. Values below use this trial-wide scale, not centimeters or each fish's individual length. Percentiles are descriptive; fish-frames are dependent and are not independent experimental replicates.

## 1. Movement: typical speed is stable under modest smoothing changes

![Movement](pilot-0066-reviewed/figures/01-movement.png)

**Claim B1-01 — derived measurement.** Median smoothed speed is **1.453 body lengths/s**; the 5th–95th percentile range is **0.355–3.531**. The speed distributions are similar across cubic 7-, 11- and 15-frame filters, compared on common eligible rows. This checks robustness over these settings, not every possible preprocessing choice.

The primary filter uses 11 frames, actual within-block timestamps and a half-window-plus-one exclusion at both block ends. Speed retains 244,940 observations (99.66%). Turn rate excludes heading-jump support and retains 244,916. Positive turning is clockwise in image coordinates. Histograms pool eligible fish-frame observations equally; the sensitivity panel displays speeds from 0 to 10 body lengths/s, while the summary table retains all finite values. There are no confidence intervals.

Acceleration and slip are available in the numerical summary but are not headline behavioral claims. An approximate residual-based acceleration noise diagnostic is about 0.44–0.48 body lengths/s² per coordinate; its independence assumptions are not established. It must not be treated as calibrated measurement uncertainty.

Evidence: [summary](pilot-0066-reviewed/summary.json), [speed sensitivity](pilot-0066-reviewed/tables/speed-sensitivity.csv), [methods](../docs/methods.md), run `pilot-0066-reviewed`.

## 2. Posture: a measurable distribution, with explicit sign conventions

![Posture](pilot-0066-reviewed/figures/02-posture.png)

**Claim B2-01 — derived geometry.** Dimensionless midpoint curvature (κ times trial-median body length) has a median of **0.015**, with 5th–95th percentiles **−0.636 to 0.694**. The distribution concentrates near straight posture, with bent configurations in both directions. This does not show that bending causes subsequent turns.

A cubic spatial fit uses seven midline points centered at point 10. All 245,780 rows are geometrically eligible. Sensitivity uses 5-, 7- and 9-point fits on the common finite set. Positive curvature is clockwise tangent rotation along the head-to-tail path; for a tailward tangent opposing body heading, this corresponds to concavity toward fish-left. It is not the positive-right-turn convention. Curvature is sensitive to pose reconstruction and the fitting neighborhood.

The three midline examples are from the lowest numeric ID at predefined 10%, 50%, 90% time quantiles (frames 2458, 12292, 22126), translated to the head and rotated into the measured body frame. They are illustrative selections, not a claim of representative posture frequency. Axes have equal length scales.

Evidence: [curvature histogram](pilot-0066-reviewed/tables/curvature.csv), [curvature sensitivity](pilot-0066-reviewed/tables/curvature-sensitivity.csv), [methods](../docs/methods.md), run `pilot-0066-reviewed`.

## 3. Organization: alignment varies, but circulation needs caution

![Organization](pilot-0066-reviewed/figures/03-organization.png)

**Claim B3-01 — derived collective measurement.** Median heading polarization is **0.894**, with repeated low-alignment episodes. Low alignment alone does not distinguish disorganization from coordinated circulation. RMS distance from the group centroid has a median of **2.844 body lengths** on eligible frames.

Time traces break at recording gaps and segment changes. Polarization uses body headings, with conservative SG-support frames around detected heading steps excluded (not all possible sustained flips) (24,554 frames retained). Spread uses smoothed positions and retains 24,494 frames. The alignment–circulation map is conditioned on circulation eligibility (46.6% of frames), which depends on relative motion and is not independent of polarization. It is not the joint distribution of the whole recording; darker bins contain more frames and do not represent an inferential probability model.

**Claim B3-02 — sensitivity result.** Centroid-relative circulation is highly sensitive to the low-speed eligibility threshold. At thresholds of 0.25, 0.5 and 1.0 body lengths/s, **93.2%, 46.6% and 2.3%** of frames qualify, respectively. The corresponding medians are 0.297, 0.383 and 0.699. These summarize different selected subsets, so the increasing median is not evidence that changing the threshold changes fish behavior.

The primary circulation threshold is 0.5 body lengths/s and requires at least 8/10 eligible fish. Its sign is saved separately. The orange reference is the approximate mean absolute score for independent isotropic directions, about 1/sqrt(pi*n); actual centroid-relative velocities are dependent. Rigid group rotation can also give C=1. It is **not** a significance threshold or a validated milling classifier. We withhold a headline claim about the fraction of time spent milling.

Evidence: [collective time series](pilot-0066-reviewed/tables/collective.csv), [threshold sensitivity](pilot-0066-reviewed/tables/threshold-sensitivity.csv), [methods](../docs/methods.md), run `pilot-0066-reviewed`.

## 4. Spacing: nearest neighbors are commonly about one body length away

![Spacing](pilot-0066-reviewed/figures/04-spacing.png)

**Claim B4-01 — descriptive spatial measurement.** Median nearest-neighbor center distance is **1.309 body lengths**, with 5th–95th percentiles **0.525–3.531**. The all-neighbor occupancy map shows a lower-density central region surrounded by more frequent neighbor locations. This could reflect geometry, spacing behavior and boundary effects; it is not a repulsion-force estimate.

The map includes all ordered focal-neighbor pairs, excluding self-pairs, across all 245,780 focal observations. Horizontal means forward; vertical means fish-right increasing downward. Bins are 0.5×0.5 body lengths within ±10 in each direction. Density is divided by all focal observations and bin area; **98.61%** of ordered pairs fall within the displayed range. The nearest-neighbor histogram uses all finite distances, including observations outside the map. Arena/wall effects have not been controlled. This occupancy map rotates by all raw headings, including heading-jump support and the unresolved reverse-motion candidates; those possible defects are disclosed rather than silently corrected.

Evidence: [spatial map data](pilot-0066-reviewed/tables/spatial-map.csv), [nearest-distance histogram](pilot-0066-reviewed/tables/nearest.csv), [methods](../docs/methods.md), run `pilot-0066-reviewed`.

## 5. Visual access: broad, changing geometric access to neighbors

![Visibility](pilot-0066-reviewed/figures/05-visibility.png)

**Claim B5-01 — geometric reconstruction.** Fish see **6.766 of 9 neighbors on average**, with median 7, and **85.08%** of directed links have a reverse link in the same frame. Visibility varies over time. These quantify the supplied ray-cast network; they do not measure attention or influence.

All tracked frames have network coverage in this trial. Degree includes zero-degree fish; 245,780 observer-frame values are counted equally. Reciprocity is pooled over directed edges, not an unweighted average of frame fractions. Angular size is distributed over all 1,662,869 edge observations, with source convention 0.36 degrees per ray. Persistence is not implemented in this pilot.

Evidence: [degree histogram](pilot-0066-reviewed/tables/degree.csv), [visibility time series](pilot-0066-reviewed/tables/visibility-time.csv), [methods](../docs/methods.md), run `pilot-0066-reviewed`.

## What this pilot motivates next

The supported narrative is: the trial contains variable individual movement, measurable body bending, episodes of differing alignment and spread, structured neighbor occupancy, and changing reconstructed visibility. The analysis also exposes a consequential choice: circulation summaries depend strongly on which low-speed observations are excluded.

Next steps are independent computational reproduction, review of flagged/low-alignment episodes at video resolution, and the same quality/measurement pipeline across the other trials before making group-size claims. Only then should response maps test whether neighbor location or visibility predicts subsequent turning. Neither leadership nor pairwise sufficiency has been established here.
